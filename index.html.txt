/**
* Web Audio API Sound Synthesizer Class
* Custom procedural arcade sound effects without external audio assets.
*/
class SoundFX {
constructor() {
this.ctx = null;
this.enabled = true;
}

init() {
if (!this.ctx) {
const AudioContext = window.AudioContext || window.webkitAudioContext;
this.ctx = new AudioContext();
}
if (this.ctx.state === 'suspended') {
this.ctx.resume();
}
}

playLaser() {
if (!this.enabled || !this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(800, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(150, this.ctx.currentTime + 0.12);

gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.12);

osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.12);
}

playExplosion(isBoss = false) {
if (!this.enabled || !this.ctx) return;
const dur = isBoss ? 0.5 : 0.25;
const bufferSize = this.ctx.sampleRate * dur;
const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
const data = buffer.getChannelData(0);
for (let i = 0; i < bufferSize; i++) {
data[i] = Math.random() * 2 - 1;
}

const noise = this.ctx.createBufferSource();
noise.buffer = buffer;

const filter = this.ctx.createBiquadFilter();
filter.type = 'lowpass';
filter.frequency.setValueAtTime(isBoss ? 400 : 800, this.ctx.currentTime);
filter.frequency.exponentialRampToValueAtTime(30, this.ctx.currentTime + dur);

const gain = this.ctx.createGain();
gain.gain.setValueAtTime(isBoss ? 0.4 : 0.25, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + dur);

noise.connect(filter);
filter.connect(gain);
gain.connect(this.ctx.destination);
noise.start();
}

playHit() {
if (!this.enabled || !this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'square';
osc.frequency.setValueAtTime(150, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(40, this.ctx.currentTime + 0.08);

gain.gain.setValueAtTime(0.3, this.ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.08);

osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.08);
}

playPowerup() {
if (!this.enabled || !this.ctx) return;
const osc = this.ctx.createOscillator();
const gain = this.ctx.createGain();
osc.type = 'sine';
osc.frequency.setValueAtTime(300, this.ctx.currentTime);
osc.frequency.exponentialRampToValueAtTime(800, this.ctx.currentTime + 0.2);

gain.gain.setValueAtTime(0.2, this.ctx.currentTime);
gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.2);

osc.connect(gain);
gain.connect(this.ctx.destination);
osc.start();
osc.stop(this.ctx.currentTime + 0.2);
}
}

/**
* Main Game Engine Architecture
*/
class GameEngine {
constructor() {
this.canvas = document.getElementById('gameCanvas');
this.ctx = this.canvas.getContext('2d');
this.width = 0;
this.height = 0;
this.centerX = 0;
this.centerY = 0;

// Game States
this.state = 'START'; // START, PLAYING, PAUSED, GAMEOVER
this.difficulty = 'EASY'; // EASY, MEDIUM, HARD
this.score = 0;
this.highScore = parseInt(localStorage.getItem('neon_strike_highscore') || '0', 10);
this.combo = 0;
this.maxCombo = 0;
this.health = 100;
this.maxHealth = 100;
this.wave = 1;
this.waveTimer = 0;

// Defender Turret Player Properties
this.player = {
angle: -Math.PI / 2,
rotationSpeed: 0.07,
radius: 28,
turretLength: 38,
cooldown: 0,
fireRate: 10 // frames per shot
};

// Object Pools & Collections
this.bullets = [];
this.enemies = [];
this.particles = [];
this.floatingTexts = [];

// Controls State
this.keys = {};
this.mousePos = { x: 0, y: 0 };
this.usingMouse = false;
this.touchFiring = false;

// Bind UI Elements
this.bindUIEvents();
this.resizeCanvas();
window.addEventListener('resize', () => this.resizeCanvas());

// Update Highscore Display
document.getElementById('high-score-display').textContent = this.highScore;

// Start Master Game Loop
this.lastTime = performance.now();
requestAnimationFrame(this.loop.bind(this));
}

resizeCanvas() {
const rect = this.canvas.parentElement.getBoundingClientRect();
this.width = rect.width;
this.height = rect.height;
this.canvas.width = this.width;
this.canvas.height = this.height;
this.centerX = this.width / 2;
this.centerY = this.height / 2;
}

bindUIEvents() {
// Keyboard Controls
window.addEventListener('keydown', (e) => {
this.keys[e.code] = true;
if (e.code === 'KeyP') this.togglePause();
if (e.code === 'Space' && this.state === 'PLAYING') {
e.preventDefault();
}
});
window.addEventListener('keyup', (e) => {
this.keys[e.code] = false;
});

// Mouse Movement & Firing
this.canvas.addEventListener('mousemove', (e) => {
const rect = this.canvas.getBoundingClientRect();
this.mousePos.x = e.clientX - rect.left;
this.mousePos.y = e.clientY - rect.top;
this.usingMouse = true;
});

this.canvas.addEventListener('mousedown', (e) => {
if (this.state === 'PLAYING') {
sfx.init();
this.shoot();
}
});

// Touch Screen Controls
this.canvas.addEventListener('touchstart', (e) => {
if (e.touches.length > 0) {
const rect = this.canvas.getBoundingClientRect();
this.mousePos.x = e.touches[0].clientX - rect.left;
this.mousePos.y = e.touches[0].clientY - rect.top;
this.usingMouse = true;
if (this.state === 'PLAYING') {
sfx.init();
this.shoot();
}
}
}, { passive: true });

// UI Screen Buttons
document.getElementById('btn-start').addEventListener('click', () => {
sfx.init();
this.startGame();
});
document.getElementById('btn-restart').addEventListener('click', () => {
sfx.init();
this.startGame();
});
document.getElementById('btn-menu').addEventListener('click', () => this.showMenu());
document.getElementById('btn-pause-menu').addEventListener('click', () => this.showMenu());
document.getElementById('btn-pause').addEventListener('click', () => this.togglePause());
document.getElementById('btn-resume').addEventListener('click', () => this.togglePause());

e.target.classList.add('bg-cyan-950/40', 'text-cyan-400', 'border-cyan-500/50');

});

// On-screen Mobile Controls
const btnLeft = document.getElementById('btn-left');
const btnRight = document.getElementById('btn-right');
const btnFire = document.getElementById('btn-fire');

const addTouchHold = (elem, keyName) => {
elem.addEventListener('touchstart', (e) => { e.preventDefault(); this.keys[keyName] = true; });
elem.addEventListener('touchend', (e) => { e.preventDefault(); this.keys[keyName] = false; });
};
addTouchHold(btnLeft, 'ArrowLeft');
addTouchHold(btnRight, 'ArrowRight');

btnFire.addEventListener('touchstart', (e) => {
e.preventDefault();
if (this.state === 'PLAYING') {
sfx.init();

}
});
}

this.score = 0;
this.combo = 0;
this.maxCombo = 0;
this.health = this.maxHealth;
this.wave = 1;
this.waveTimer = 0;
this.bullets = [];
this.enemies = [];
this.particles = [];
this.floatingTexts = [];

// Hide overlays, show HUD
document.getElementById('screen-start').classList.add('hidden');
document.getElementById('screen-gameover').classList.add('hidden');

}

showMenu() {

document.getElementById('screen-start').classList.remove('hidden');
document.getElementById('screen-gameover').classList.add('hidden');
document.getElementById('screen-pause').classList.add('hidden');

togglePause() {
if (this.state === 'PLAYING') {
this.state = 'PAUSED';
document.getElementById('screen-pause').classList.remove('hidden');
} else if (this.state === 'PAUSED') {
this.state = 'PLAYING';
document.getElementById('screen-pause').classList.add('hidden');
}
}

gameOver() {
this.state = 'GAMEOVER';
if (this.score > this.highScore) {
this.highScore = this.score;
localStorage.setItem('neon_strike_highscore', this.highScore.toString());
}

document.getElementById('go-score').textContent = this.score;
document.getElementById('go-combo').textContent = `x${this.maxCombo}`;
document.getElementById('go-wave').textContent = this.wave;
document.getElementById('go-highscore').textContent = this.highScore;

document.getElementById('screen-gameover').classList.remove('hidden');
document.getElementById('hud').classList.add('opacity-0');
document.getElementById('touch-controls').classList.add('opacity-0');
}

shoot() {
if (this.player.cooldown > 0) return;

// Fire projectile along turret angle
const speed = 12;
const tipX = this.centerX + Math.cos(this.player.angle) * this.player.turretLength;
const tipY = this.centerY + Math.sin(this.player.angle) * this.player.turretLength;

this.bullets.push({
x: tipX,
y: tipY,
vx: Math.cos(this.player.angle) * speed,
vy: Math.sin(this.player.angle) * speed,
radius: 4,
color: '#06b6d4'
});

// Muzzle flash particles
for (let i = 0; i < 5; i++) {
const pAngle = this.player.angle + (Math.random() - 0.5) * 0.4;
const pSpeed = Math.random() * 4 + 2;
this.particles.push({
x: tipX,
y: tipY,
vx: Math.cos(pAngle) * pSpeed,
vy: Math.sin(pAngle) * pSpeed,
life: 1,
decay: Math.random() * 0.1 + 0.05,
color: '#38bdf8',
size: Math.random() * 3 + 1
});
}

sfx.playLaser();
this.player.cooldown = this.player.fireRate;
}

spawnEnemy() {
// Spawn enemy from edges toward the center core
const angle = Math.random() * Math.PI * 2;
const spawnDistance = Math.max(this.width, this.height) / 2 + 40;
const x = this.centerX + Math.cos(angle) * spawnDistance;
const y = this.centerY + Math.sin(angle) * spawnDistance;

// Difficulty multipliers
let speedMult = 1;
let hpMult = 1;
if (this.difficulty === 'MEDIUM') { speedMult = 1.25; hpMult = 1.3; }
if (this.difficulty === 'HARD') { speedMult = 1.5; hpMult = 1.7; }

// Enemy Type Randomization
const rand = Math.random();
let type = 'STANDARD';
let radius = 14;
let speed = (Math.random() * 0.8 + 1.2) * speedMult;
let hp = Math.floor((1 + Math.floor(this.wave / 3)) * hpMult);
let scoreVal = 100;
let color = '#ec4899'; // Neon Pink

if (rand < 0.3) {
// Fast Scout Drone
type = 'FAST';
radius = 10;
speed = (Math.random() * 1.2 + 2.2) * speedMult;
hp = 1;
scoreVal = 150;
color = '#f59e0b'; // Neon Amber
} else if (rand > 0.85 && this.wave >= 2) {
// Armored Heavy Drone / Boss
type = 'HEAVY';
radius = 22;
speed = (Math.random() * 0.4 + 0.7) * speedMult;
hp = Math.floor((4 + Math.floor(this.wave / 2)) * hpMult);
scoreVal = 350;
color = '#a855f7'; // Neon Purple
}

this.enemies.push({
x, y,
type,
radius,
speed,
hp,
maxHp: hp,
scoreVal,
color,
pulse: Math.random() * Math.PI
});
}

spawnFloatingText(text, x, y, color = '#38bdf8') {
this.floatingTexts.push({
text,
x, y,
vy: -1.5,
life: 1,
decay: 0.02,
color
});
}

updateHUD() {
document.getElementById('hud-score').textContent = this.score;
document.getElementById('hud-wave').textContent = `WAVE ${this.wave}`;

const healthBar = document.getElementById('health-bar');
const pct = Math.max(0, (this.health / this.maxHealth) * 100);
healthBar.style.width = `${pct}%`;

if (pct < 30) {
healthBar.className = "h-full bg-gradient-to-r from-rose-600 to-red-500 rounded-full transition-all duration-300 w-full animate-pulse";
} else {
healthBar.className = "h-full bg-gradient-to-r from-cyan-500 to-emerald-400 rounded-full transition-all duration-300 w-full";
}

// Combo HUD
const comboElem = document.getElementById('hud-combo');
if (this.combo > 1) {
document.getElementById('combo-multiplier').textContent = this.combo;
comboElem.classList.remove('opacity-0', 'scale-100');
comboElem.classList.add('opacity-100', 'scale-110');
} else {
comboElem.classList.add('opacity-0');
comboElem.classList.remove('opacity-100', 'scale-110');
}
}

update(delta) {
if (this.state !== 'PLAYING') return;

// Player Rotation Control
if (this.keys['KeyA'] || this.keys['ArrowLeft']) {
this.player.angle -= this.player.rotationSpeed;
this.usingMouse = false;
}
if (this.keys['KeyD'] || this.keys['ArrowRight']) {
this.player.angle += this.player.rotationSpeed;
this.usingMouse = false;
}

// Keyboard Firing
if (this.keys['Space']) {
this.shoot();
}

// Mouse / Touch Aiming
if (this.usingMouse) {
const dx = this.mousePos.x - this.centerX;
const dy = this.mousePos.y - this.centerY;
this.player.angle = Math.atan2(dy, dx);
}

if (this.player.cooldown > 0) this.player.cooldown--;

// Wave System & Spawning Logic
this.waveTimer++;
const spawnInterval = Math.max(20, 90 - (this.wave * 8));
if (this.waveTimer % Math.floor(spawnInterval) === 0) {
this.spawnEnemy();
}
if (this.waveTimer > 1200) { // Increment Wave every ~20s
this.wave++;
this.waveTimer = 0;
this.spawnFloatingText(`WAVE ${this.wave} STARTED!`, this.centerX, this.centerY - 60, '#e0e7ff');
sfx.playPowerup();
}

// Update Projectiles
for (let i = this.bullets.length - 1; i >= 0; i--) {
const b = this.bullets[i];
b.x += b.vx;
b.y += b.vy;

// Screen boundary removal
if (b.x < 0 || b.x > this.width || b.y < 0 || b.y > this.height) {
this.bullets.splice(i, 1);
}
}

// Update Enemies
for (let i = this.enemies.length - 1; i >= 0; i--) {
const e = this.enemies[i];
e.pulse += 0.05;

// Vector towards core center
const dx = this.centerX - e.x;
const dy = this.centerY - e.y;
const dist = Math.hypot(dx, dy);

e.x += (dx / dist) * e.speed;
e.y += (dy / dist) * e.speed;

// Core Collision Check
if (dist < this.player.radius + e.radius) {
this.health -= (e.type === 'HEAVY' ? 25 : 10);
this.combo = 0; // Reset Combo
this.updateHUD();
sfx.playHit();

// Explosion FX on Core
for (let p = 0; p < 12; p++) {
const pAng = Math.random() * Math.PI * 2;
const pSpd = Math.random() * 5 + 1;
this.particles.push({
x: e.x, y: e.y,
vx: Math.cos(pAng) * pSpd, vy: Math.sin(pAng) * pSpd,
life: 1, decay: 0.04, color: '#f43f5e', size: Math.random() * 4 + 2
});
}

this.enemies.splice(i, 1);

if (this.health <= 0) {
this.gameOver();
}
continue;
}

// Bullet Collision Check
for (let j = this.bullets.length - 1; j >= 0; j--) {
const b = this.bullets[j];
const bDist = Math.hypot(b.x - e.x, b.y - e.y);

if (bDist < e.radius + b.radius) {
e.hp--;
this.bullets.splice(j, 1);

// Hit particle spark
for (let p = 0; p < 4; p++) {
this.particles.push({
x: b.x, y: b.y,
vx: (Math.random() - 0.5) * 4, vy: (Math.random() - 0.5) * 4,
life: 1, decay: 0.1, color: '#38bdf8', size: 2
});
}

if (e.hp <= 0) {
// Enemy Destroyed
this.combo++;
if (this.combo > this.maxCombo) this.maxCombo = this.combo;
const pts = e.scoreVal * Math.min(this.combo, 10);
this.score += pts;
this.updateHUD();

sfx.playExplosion(e.type === 'HEAVY');
this.spawnFloatingText(`+${pts}`, e.x, e.y, e.color);

// Explosion FX
for (let p = 0; p < (e.type === 'HEAVY' ? 24 : 12); p++) {
const pAng = Math.random() * Math.PI * 2;
const pSpd = Math.random() * (e.type === 'HEAVY' ? 6 : 4) + 1;
this.particles.push({
x: e.x, y: e.y,
vx: Math.cos(pAng) * pSpd, vy: Math.sin(pAng) * pSpd,
life: 1, decay: Math.random() * 0.03 + 0.02,
color: e.color, size: Math.random() * 4 + 2
});
}

this.enemies.splice(i, 1);
break;
} else {
sfx.playHit();
}
}
}
}

// Update Particle FX
for (let i = this.particles.length - 1; i >= 0; i--) {
const p = this.particles[i];
p.x += p.vx;
p.y += p.vy;
p.life -= p.decay;
if (p.life <= 0) this.particles.splice(i, 1);
}

// Update Floating Text FX
for (let i = this.floatingTexts.length - 1; i >= 0; i--) {
const ft = this.floatingTexts[i];
ft.y += ft.vy;
ft.life -= ft.decay;
if (ft.life <= 0) this.floatingTexts.splice(i, 1);
}
}

render() {
// Clear Canvas background
this.ctx.fillStyle = '#030712';
this.ctx.fillRect(0, 0, this.width, this.height);

// Render Radial Background Grid / Orbits
this.ctx.save();
this.ctx.strokeStyle = 'rgba(15, 23, 42, 0.6)';
this.ctx.lineWidth = 1;
for (let r = 100; r < Math.max(this.width, this.height); r += 100) {
this.ctx.beginPath();
this.ctx.arc(this.centerX, this.centerY, r, 0, Math.PI * 2);
this.ctx.stroke();
}
this.ctx.restore();

// Render Central Defender Base Core
this.ctx.save();
// Outer Glow Ring
this.ctx.shadowBlur = 15;
this.ctx.shadowColor = '#06b6d4';
this.ctx.strokeStyle = '#06b6d4';
this.ctx.lineWidth = 3;
this.ctx.beginPath();
this.ctx.arc(this.centerX, this.centerY, this.player.radius, 0, Math.PI * 2);
this.ctx.stroke();

// Inner Core Fill
this.ctx.fillStyle = '#0891b2';
this.ctx.beginPath();
this.ctx.arc(this.centerX, this.centerY, this.player.radius - 6, 0, Math.PI * 2);
this.ctx.fill();

// Turret Barrel
this.ctx.translate(this.centerX, this.centerY);
this.ctx.rotate(this.player.angle);
this.ctx.fillStyle = '#38bdf8';
this.ctx.fillRect(0, -5, this.player.turretLength, 10);
this.ctx.restore();

// Render Projectiles
for (const b of this.bullets) {
this.ctx.save();
this.ctx.shadowBlur = 10;
this.ctx.shadowColor = b.color;
this.ctx.fillStyle = b.color;
this.ctx.beginPath();
this.ctx.arc(b.x, b.y, b.radius, 0, Math.PI * 2);
this.ctx.fill();
this.ctx.restore();
}

// Render Enemies
for (const e of this.enemies) {
this.ctx.save();
this.ctx.shadowBlur = 12;
this.ctx.shadowColor = e.color;
this.ctx.strokeStyle = e.color;
this.ctx.lineWidth = 2.5;

const pSize = e.radius + Math.sin(e.pulse) * 2;

if (e.type === 'STANDARD') {
// Diamond Shape
this.ctx.beginPath();
this.ctx.moveTo(e.x, e.y - pSize);
this.ctx.lineTo(e.x + pSize, e.y);
this.ctx.lineTo(e.x, e.y + pSize);
this.ctx.lineTo(e.x - pSize, e.y);
this.ctx.closePath();
this.ctx.stroke();
} else if (e.type === 'FAST') {
// Triangle Arrow
const angle = Math.atan2(this.centerY - e.y, this.centerX - e.x);
this.ctx.translate(e.x, e.y);
this.ctx.rotate(angle);
this.ctx.beginPath();
this.ctx.moveTo(pSize, 0);
this.ctx.lineTo(-pSize, -pSize / 1.5);
this.ctx.lineTo(-pSize, pSize / 1.5);
this.ctx.closePath();
this.ctx.stroke();
this.ctx.rotate(-angle);
this.ctx.translate(-e.x, -e.y);
} else if (e.type === 'HEAVY') {
// Octagon Boss / Heavy Drone
this.ctx.beginPath();
for (let a = 0; a < 6; a++) {
const rad = (Math.PI / 3) * a;
const px = e.x + Math.cos(rad) * pSize;
const py = e.y + Math.sin(rad) * pSize;
if (a === 0) this.ctx.moveTo(px, py);
else this.ctx.lineTo(px, py);
}
this.ctx.closePath();
this.ctx.stroke();

// HP Bar for Heavy Drone
if (e.hp < e.maxHp) {
const barW = e.radius * 2;
this.ctx.fillStyle = 'rgba(15, 23, 42, 0.8)';
this.ctx.fillRect(e.x - e.radius, e.y - e.radius - 12, barW, 4);
this.ctx.fillStyle = e.color;
this.ctx.fillRect(e.x - e.radius, e.y - e.radius - 12, barW * (e.hp / e.maxHp), 4);
}
}

this.ctx.restore();
}

// Render Particles
for (const p of this.particles) {
this.ctx.save();
this.ctx.globalAlpha = Math.max(0, p.life);
this.ctx.fillStyle = p.color;
this.ctx.beginPath();
this.ctx.arc(p.x, p.y, p.size, 0, Math.PI * 2);
this.ctx.fill();
this.ctx.restore();
}

// Render Floating Score Text
for (const ft of this.floatingTexts) {
this.ctx.save();
this.ctx.globalAlpha = Math.max(0, ft.life);
this.ctx.fillStyle = ft.color;
this.ctx.font = 'bold 14px Orbitron';
this.ctx.textAlign = 'center';
this.ctx.fillText(ft.text, ft.x, ft.y);
this.ctx.restore();
}
}

loop(timestamp) {
const delta = timestamp - this.lastTime;
this.lastTime = timestamp;

this.update(delta);
this.render();

requestAnimationFrame(this.loop.bind(this));
}
}

// Initialize Game Engine on Load
window.addEventListener('load', () => {
window.game = new GameEngine();
});
